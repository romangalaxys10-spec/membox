    52→    52→function CopyButton({ text, label }: { text: string; label?: string }) {

    60→    60→    <Button

    62→    62→      size="sm"

    64→    64→      className="h-7 gap-1.5 text-xs"

    69→    69→  )

    72→    72→function formatBytes(bytes: number): string {
    73→    73→  if (bytes === 0) return '0 B'
    74→    74→  const k = 1024
    75→    75→  const sizes = ['B', 'KB', 'MB', 'GB']
    76→    76→  const i = Math.floor(Math.log(bytes) / Math.log(k))
    77→    77→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i]
    78→    78→}
