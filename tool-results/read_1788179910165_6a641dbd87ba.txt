    83→function formatBytes(bytes: number): string {
    84→  if (bytes === 0) return '0 B'
    85→  const k = 1024
    86→  const sizes = ['B', 'KB', 'MB', 'GB']
    87→  const i = Math.floor(Math.log(bytes) / Math.log(k))
    88→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i]
    89→}

   105→function getFileIcon(name: string) {
   106→  const ext = name.split('.').pop()?.toLowerCase() || ''
   107→  const docExts = ['pdf', 'doc', 'docx', 'odt', 'rtf', 'txt', 'md', 'epub', 'mobi']
   108→  const sheetExts = ['xls', 'xlsx', 'ods', 'csv', 'tsv']
   109→  const imgExts = ['png', 'jpg', 'jpeg', 'gif', 'webp', 'svg', 'ico', 'bmp', 'avif']
   110→  const codeExts = ['js', 'ts', 'jsx', 'tsx', 'py', 'rb', 'go', 'rs', 'java', 'c', 'cpp', 'h', 'css', 'html', 'json', 'yaml', 'yml', 'xml', 'sql', 'sh', 'vue', 'svelte']
   111→  const archiveExts = ['zip', 'tar', 'gz', 'bz2', '7z', 'rar']
   112→  const mediaExts = ['mp3', 'wav', 'ogg', 'flac', 'mp4', 'webm', 'avi', 'mov', 'mkv']
   113→
   114→  if (docExts.includes(ext)) return <FileText className="h-4 w-4 text-blue-400" />
   115→  if (sheetExts.includes(ext)) return <FileSpreadsheet className="h-4 w-4 text-emerald-400" />
   116→  if (imgExts.includes(ext)) return <FileImage className="h-4 w-4 text-purple-400" />
   117→  if (codeExts.includes(ext)) return <FileCode className="h-4 w-4 text-amber-400" />
   118→  if (archiveExts.includes(ext)) return <FileArchive className="h-4 w-4 text-orange-400" />
   119→  if (mediaExts.includes(ext)) return <File className="h-4 w-4 text-pink-400" />
   120→  return <File className="h-4 w-4 text-zinc-400" />
   121→}

   123→function FileTreeItem({
   124→  file,
   125→  slug,
   126→  token,
   127→  basePath,
   128→}: {
   129→  file: { name: string; type: 'file' | 'directory'; size?: number; modified?: string }
   130→  slug: string
   131→  token: string
   132→  basePath: string
   133→}) {
   134→  const downloadUrl = `${basePath}/api/m/${slug}/files/${name}`
   135→  const headers = new Headers()
   136→  headers.set('Authorization', `Bearer ${token}`)
   137→
   138→  return (
   139→    <div className="flex items-center gap-2 px-3 py-1.5 rounded-md hover:bg-zinc-800/50 text-sm group">
   140→      {file.type === 'directory' ? (
   141→        <Folder className="h-4 w-4 text-amber-400 shrink-0" />
   142→      ) : (
   143→        <div className="shrink-0">{getFileIcon(file.name)}</div>
   144→      )}
   145→      <span className="flex-1 text-zinc-300 font-mono text-xs truncate" title={file.name}>
   146→        {file.name}
   147→      </span>
   148→      {file.type === 'file' && file.size !== undefined && (
   149→        <span className="text-[11px] text-zinc-500 shrink-0">{formatBytes(file.size!)}</span>
   150→      )}
   151→      {file.modified && (
   152→        <span className="text-[11px] text-zinc-600 shrink-0 hidden sm:inline">
   153→          {new Date(file.modified).toLocaleDateString()}
   154→        </span>
   155→      )}
   156→      {file.type === 'file' && (
   157→        <a
   158→          href={`/api/m/${slug}/files/${file.name}?token=${token}`}
   159→          className="opacity-0 group-hover:opacity-100 transition-opacity shrink-0"
   160→          title="Download"
   161→        >
   162→          <Download className="h-3.5 w-3.5 text-zinc-400 hover:text-emerald-400" />
   163→        </a>
   164→      )}
   165→    </div>
   166→  )
   167→}

   169→function StorageTree({ files, slug, token }: { files: MemBox['files']; slug: string; token: string }) {
   170→  if (!files || files.length === 0) {

   172→  }

   174→  return (
   175→    <div className="space-y-0.5">

   177→        <FileTreeItem key={f.name} file={f} slug={slug} token={token} basePath={basePath} />
   178→      ))}
   179→    </div>
   180→  )
