    52→function CopyButton({ text, label }: { text: string; label?: string }) {
    53→  const [copied, setCopied] = useState(false)
    54→  const handleCopy = async () => {
    55→    await navigator.clipboard.writeText(text)
    56→    setCopied(true)
    57→    setTimeout(() => setCopied(false), 2000)
    58→  }
    59→  return (
    60→    <Button
    61→      variant="ghost"
    62→      size="sm"
    63→      onClick={handleCopy}
    64→      className="h-7 gap-1.5 text-xs"
    65→    >
    66→      {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
    67→      {label || (copied ? 'Copied' : 'Copy')}
    68→    </Button>
    69→  )
    70→}

    72→function formatBytes(bytes: number): string {
    73→  if (bytes === 0) return '0 B'
    74→  const k = 1024
    75→  const sizes = ['B', 'KB', 'MB', 'GB']
    76→  const i = Math.floor(Math.log(bytes) / Math.log(k))
    77→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i]
    78→}

    94→function StorageTree({ files, slug, token }: { files: MemBox['files']; slug: string; token: string }) {
    95→  if (!files || files.length === 0) {
    96→    return <p className="text-sm text-zinc-500 py-4 text-center">No files yet. Use the API to store memories.</p>
    97→  }
    98→  return (
    99→    <div className="space-y-1">
   100→      {files.map((f) => (
   101→        <div key={f.name} className="flex items-center gap-2 px-3 py-1.5 rounded-md hover:bg-zinc-800/50 text-sm">
   102→          {f.type === 'directory' ? (
   103→            <Folder className="h-4 w-4 text-amber-400" />
   104→          ) : (
   105→            <FileText className="h-4 w-4 text-zinc-400" />
   106→          )}
   107→          <span className="flex-1 text-zinc-300 font-mono text-xs">{f.name}</span>
   108→          {f.type === 'file' && f.size !== undefined && (
   109→            <span className="text-xs text-zinc-500">{formatBytes(f.size!)}</span>
   110→          )}
   111→          {f.modified && (
   112→            <span className="text-xs text-zinc-600">
   113→              {new Date(f.modified).toLocaleDateString()}
   114→            </span>
   115→          )}
   116→        </div>
   117→      ))}
   118→    </div>
   119→  )
   120→}
