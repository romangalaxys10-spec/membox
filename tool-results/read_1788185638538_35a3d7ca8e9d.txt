    48→    48→function formatBytes(b: number): string {
    49→    49→  if (b === 0) return '0 B'
    50→    50→  const k = 1024, s = ['B', 'KB', 'MB', 'GB'], i = Math.floor(Math.log(b) / Math.log(k))
    51→    51→  return parseFloat((b / Math.pow(k, i)).toFixed(1)) + ' ' + s[i]
    52→    52→}

    68→    68→function getFileIcon(name: string) {
    69→    69→  const ext = name.split('.').pop()?.toLowerCase() || ''
    70→    70→  if (['pdf','doc','docx','odt','rtf','txt','md','epub'].includes(ext)) return <FileText className="h-4 w-4 text-blue-400" />
    71→    71→  if (['xls','xlsx','ods','csv','tsv'].includes(ext)) return <FileSpreadsheet className="h-4 w-4 text-emerald-400" />
    72→    72→  if (['png','jpg','jpeg','gif','webp','svg'].includes(ext)) return <FileImage className="h-4 w-4 text-purple-400" />
    73→    73→  if (['js','ts','jsx','tsx','py','go','rs','java','c','cpp','css','html','json','yaml','sql','sh'].includes(ext)) return <FileCode className="h-4 w-4 text-amber-400" />
    74→    74→  if (['zip','tar','gz','7z','rar'].includes(ext)) return <FileArchive className="h-4 w-4 text-orange-400" />
    75→    75→  return <File className="h-4 w-4 text-zinc-500" />
