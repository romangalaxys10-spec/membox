    83→    83→function formatBytes(bytes: number): string {
    84→    84→  if (bytes === 0) return '0 B'
    85→    85→  const k = 1024
    86→    86→  const sizes = ['B', 'KB', 'MB', 'GB']
    87→    87→  const i = Math.floor(Math.log(bytes) / Math.log(k))
    88→    88→  return parseFloat((bytes / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i]
    89→    89→}

   105→   105→function getFileIcon(name: string) {
   106→   106→  const ext = name.split('.').pop()?.toLowerCase() || ''
   107→   107→  const docExts = ['pdf', 'doc', 'docx', 'odt', 'rtf', 'txt', 'md', 'epub', 'mobi']
   108→   108→  const sheetExts = ['xls', 'xlsx', 'ods', 'csv', 'tsv']
   109→   109→  const imgExts = ['png', 'jpg', 'jpeg', 'gif', 'webp', 'svg', 'ico', 'bmp', 'avif']
   110→   110→  const codeExts = ['js', 'ts', 'jsx', 'tsx', 'py', 'rb', 'go', 'rs', 'java', 'c', 'cpp', 'h', 'css', 'html', 'json', 'yaml', 'yml', 'xml', 'sql', 'sh', 'vue', 'svelte']
   111→   111→  const archiveExts = ['zip', 'tar', 'gz', 'bz2', '7z', 'rar']
   112→   112→  const mediaExts = ['mp3', 'wav', 'ogg', 'flac', 'mp4', 'webm', 'avi', 'mov', 'mkv']
   113→   113→
   114→   114→  if (docExts.includes(ext)) return <FileText className="h-4 w-4 text-blue-400" />
   115→   115→  if (sheetExts.includes(ext)) return <FileSpreadsheet className="h-4 w-4 text-emerald-400" />
   116→   116→  if (imgExts.includes(ext)) return <FileImage className="h-4 w-4 text-purple-400" />
   117→   117→  if (codeExts.includes(ext)) return <FileCode className="h-4 w-4 text-amber-400" />
   118→   118→  if (archiveExts.includes(ext)) return <FileArchive className="h-4 w-4 text-orange-400" />
   119→   119→  if (mediaExts.includes(ext)) return <File className="h-4 w-4 text-pink-400" />
   120→   120→  return <File className="h-4 w-4 text-zinc-400" />
   121→   121→}
